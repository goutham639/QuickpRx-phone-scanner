Add new HTTPS server block after PHARMACIST PORTAL block, following same pattern:

```nginx
# ----------------------------------------------------------------------
# PHONE SCANNER (HTTPS)
# ----------------------------------------------------------------------
server {
  listen 443 ssl;
  server_name scanner.quickprx.com;

  ssl_certificate     /etc/nginx/ssl/fullchain.pem;
  ssl_certificate_key /etc/nginx/ssl/privkey.pem;

  # Block common backup/dump file patterns (case-insensitive)
  location ~* \.(sql|sqlite|db|bak|backup|gz|zip|7z|tar|tgz|rar|env|log|csv)$ {
    return 404;
  }

  # Block backup path prefix
  location /backup {
    return 404;
  }

  location / {
    set $scanner_target http://phone-scanner:80;
    proxy_pass         $scanner_target;
    proxy_set_header   Host $host;
    proxy_set_header   X-Request-Id $final_request_id;
    proxy_set_header   X-Real-IP $remote_addr;
    proxy_set_header   X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header   X-Forwarded-Proto $scheme;

    add_header X-Request-Id $final_request_id always;
  }
}
```
